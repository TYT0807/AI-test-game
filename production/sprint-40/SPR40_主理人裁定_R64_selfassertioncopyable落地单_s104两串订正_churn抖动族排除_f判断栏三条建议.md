def gen_triple_ok(relpath):
    """代次绑定三元组现核（R64 §C / R63 §E）：返回 (ok, why, 现读三元组)。
    宿主不可读 ⇒ False（缺宿主＝无法证代次，⛔ 放行）。"""
    import hashlib
    p = relpath if os.path.isabs(relpath) else os.path.join(ROOT, relpath)
    try:
        with open(p, "rb") as fh:
            blob = fh.read()
    except OSError as e:
        return False, "宿主不可读(%s)" % e.__class__.__name__, "-"
    nlines = blob.count(b"\n")
    if not blob.endswith(b"\n"):
        nlines += 1
    return True, "", "%s@%s/%d" % (relpath, hashlib.md5(blob).hexdigest()[:8], nlines)
